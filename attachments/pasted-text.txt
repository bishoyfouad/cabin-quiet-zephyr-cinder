  function evaluateLeaf(v) {
    if (typeof v === 'string') {
      v = v.toLowerCase();
      const slash = v.lastIndexOf('/');
      if (slash !== -1) v = v.slice(slash + 1);
    }
    if (v === false || v === 0 || v === '0' || v === 'false' || v === 'outofstock' || v === 'out_of_stock') {
      return true;
    }
    if (v === true || v === 'true' || v === 'instock' || v === 'in_stock' || (typeof v === 'number' && v > 0)) {
      return false;
    }
    return null;
  }

  function findAvailFlag(node, depth) {
    if (depth === undefined) depth = 0;
    if (depth > 3 || node == null || typeof node !== 'object') return null;

    if (Array.isArray(node)) {
      let sawValidSignal = false;
      for (let i = 0; i < node.length; i++) {
        const res = findAvailFlag(node[i], depth + 1);
        if (res === false) return false;
        if (res === true) sawValidSignal = true;
      }
      return sawValidSignal ? true : null;
    }

    for (const k in node) {
      if (AVAIL_KEYS.test(k)) {
        const res = evaluateLeaf(node[k]);
        if (res !== null) return res;
      }
      if (STOCK_QTY_KEYS.test(k) && typeof node[k] === 'number') {
        return node[k] <= 0;
      }
    }

    if (node.variants && Array.isArray(node.variants)) {
      const res = findAvailFlag(node.variants, depth + 1);
      if (res !== null) return res;
    }

    for (const k in node) {
      if (k === 'variants') continue;
      const res = findAvailFlag(node[k], depth + 1);
      if (res !== null) return res;
    }

    return null;
  }

    }

  }

  function classifyAttrs(el, isTitle) {
    let oos = 0, instock = 0, why = '';
    const attrs = el.attributes;
    if (!attrs) return { oos: 0, instock: 0, why: '' };
    for (let i = 0; i < attrs.length; i++) {
      const a = attrs[i];
      const n = String(a.name || '').toLowerCase();
      const v = String(a.value || '');
      const raw = v.trim();
      const kv = KEY(v);

      /* [v5.3.2] explicit inventory-status attributes get hard signal */
      if (n === 'data-inventory-status' || n === 'data-stock-status') {
        if (/^(instock|lowstock|limitedstock|available|preorder|backorder)$/.test(kv)) { instock = Math.max(instock, 4); continue; }
        if (/^(outofstock|soldout|unavailable|nostock)$/.test(kv)) { oos = Math.max(oos, 4); why = why || '@' + a.name; continue; }
      }

      if (n === 'class' || n === 'id') {
        const tokens = raw.toLowerCase().split(/\s+/);
        for (let t = 0; t < tokens.length; t++) {
          const tok = tokens[t];
          if (!tok) continue;
          if (HARD_INSTOCK_TOKEN.test(tok)) {
            if (instock < 4) instock = 4;
            continue;
          }
          if (HARD_OOS_TOKEN.test(tok)) {
            if (oos < 4) { oos = 4; if (!why) why = n + '="' + tok + '"'; }
            continue;
          }
          const k = KEY(tok);
          if (!k) continue;
          if (OOS_TOKENS.has(k)) {
            if (oos < 3) { oos = 3; if (!why) why = n + '="' + tok + '"'; }
          } else if (STOCK_TOKENS.has(k)) {
            if (instock < 3) instock = 3;
          } else if (OOS_RE.test(k)) {
            if (oos < 3) { oos = 3; if (!why) why = n + '="' + tok + '"'; }
          } else if (STOCK_RE.test(k)) {
            if (instock < 2) instock = 2;
          }
        }
        continue;
      }
      if (n === 'style') {
        if (/grayscale\s*\(\s*(0?\.[5-9]|1)/i.test(v) || /opacity\s*:\s*0?\.[0-8]\b/i.test(v)) { if (!oos) { oos = 1; why = 'inline style'; } }
        continue;
      }
      if (n === 'href' || n === 'action') {
        if (/schema\.org/i.test(v)) {
          if (/schema\.org\/?InStock/i.test(v)) { instock = Math.max(instock, 4); continue; }
          if (OOS_RE.test(kv) || OOS_TOKENS.has(kv)) {
            if (el.getClientRects && el.getClientRects().length) { oos = 3; why = 'schema.org link'; }
          }
        }
        continue;
      }
      if (isTitle && (n === 'alt' || n === 'title' || n === 'aria-label')) continue;
      if (n === 'src' || n === 'srcset' || n === 'data-src' || n === 'data-srcset' ||
          n === 'data-original' || n === 'data-lazy-src' || n === 'poster') {
        if (OOS_RE.test(kv) || OOS_TOKENS.has(kv)) { if (!oos) { oos = 1; why = 'image filename'; } }
        continue;
      }
      const rl = raw.toLowerCase();
      if (/^data-(is-)?(sold-?out|out-?of-?stock|oos|unavailable)$/.test(n) && POSVAL.test(rl)) { oos = Math.max(oos, 3); why = why || '@' + a.name; continue; }
      if (/^data-(is-)?(available|in-?stock|stock|inventory(-?quantity)?)$/.test(n) && /^(0|false|no)$/.test(rl)) { oos = Math.max(oos, 3); why = why || '@' + a.name; continue; }
      if (NEGVAL.test(raw)) continue;
      if (n === 'alt' || n === 'title' || n === 'aria-label') {
        if (OOS_RE.test(kv) || OOS_TOKENS.has(kv)) { if (!oos) { oos = raw.length <= 24 ? 3 : 1; why = '@' + a.name; } }
        continue;
      }
      if (OOS_RE.test(kv) || OOS_TOKENS.has(kv)) {
        if (NAME_OOS.test(n) && POSVAL.test(raw)) { oos = Math.max(oos, 3); why = why || '@' + a.name; }
        else if (NAME_OOS.test(n) && !POSVAL.test(raw) && kv.length <= 10) { /* ambiguous */ }
        else { oos = Math.max(oos, 3); why = why || '@' + a.name; }
        continue;
      }
      if (NAME_AVAIL.test(n) && (kv === '0' || kv === 'false' || kv === 'no')) { oos = Math.max(oos, 3); why = why || '@' + a.name + '=0'; continue; }
      if (NAME_AVAIL.test(n) && (POSVAL.test(raw) || kv === '1' || kv === 'true')) { instock = Math.max(instock, 3); continue; }
      if (n.indexOf('data-') === 0 || n.indexOf('aria-') === 0 || n === 'itemprop' || n === 'content' ||
          n === 'value' || n === 'placeholder' || n === 'name' || n === 'role') {
        if (STOCK_RE.test(kv) || STOCK_TOKENS.has(kv)) instock = Math.max(instock, 2);
      }
    }
    return { oos: oos, instock: instock, why: why };
  }

  function scanTextBadges(card, titleEl) {
    const els = card.querySelectorAll('span,div,p,b,strong,em,i,small,label,button,a,output,li,dd,dt,h4,h5,h6,figcaption,caption,td');
    for (let i = 0; i < els.length; i++) {
      const el = els[i];
      if (titleEl && (el === titleEl || el.contains(titleEl))) continue;
      if (el.children.length > 4) continue;
      if (typeof el.getClientRects === 'function' && !el.getClientRects().length) continue;
      const t = textOf(el);
      if (!t || t.length > 60) continue;
      if (el.querySelector('a[href], h1, h2, h3, h4, h5, img, [class*="price" i]')) continue;
      const k = KEY(t).replace(/منفذ|منافذ/g, '');
      if (!k || k.length > 46) continue;
      if (OOS_TOKENS.has(k) || OOS_RE.test(k)) return t.slice(0, 40);
    }
    return '';
  }

  function scanBuy(card) {
    const out = { present: false, count: 0, on: 0, off: 0, oosLabel: false };
    const els = card.querySelectorAll('a,button,[role="button"],input[type="submit"]');
    for (let i = 0; i < els.length; i++) {
      const el = els[i];
      const label = KEY([el.getAttribute('aria-label') || '', el.getAttribute('title') || '',
        el.getAttribute('data-action') || '', (typeof el.className === 'string' ? el.className : ''),
        el.getAttribute('value') || '', textOf(el)].join(' '));
      if (!label) continue;
      const oosHit = OOS_RE.test(label) || OOS_TOKENS.has(label);
      const isBuy = BUY_RE.test(label);
      if (oosHit) { out.oosLabel = true; continue; }
      if (!isBuy) continue;
      out.present = true; out.count++;
      if (isDisabledEl(el)) out.off++; else out.on++;
    }
    return out;
  }

  function computeEvidence(card, titleEl) {
    const ev = { score: 0, hardOos: false, hardInStock: false, why: [], tokens: [], feats: {} };
    if (!card || card.nodeType !== 1) return ev;

    const jsonSignal = searchJsonState(card);
    if (jsonSignal === true) {
      ev.score += 5;
      ev.hardOos = true;
      ev.why.push('json attribute state (available: false)');
      return ev;
    } else if (jsonSignal === false) {
      ev.hardInStock = true;
      return ev;
    }

    const region = oosRegion(card);
    const els = []; const seen = new Set();
    function add(el) { if (el && el.nodeType === 1 && !seen.has(el)) { seen.add(el); els.push(el); } }
    add(region); add(card);
    const all = card.querySelectorAll('*');
    for (let i = 0; i < all.length && els.length < 240; i++) add(all[i]);
    if (region !== card) { const rc = region.children; for (let i = 0; i < rc.length; i++) add(rc[i]); }
    const toks = new Set();
    for (let i = 0; i < els.length; i++) {
      const cls = (typeof els[i].className === 'string' ? els[i].className : (els[i].getAttribute && els[i].getAttribute('class')) || '');
      const parts = String(cls).split(/\s+/);
      for (let j = 0; j < parts.length; j++) if (parts[j] && parts[j].length < 40) toks.add(parts[j]);
    }
    ev.tokens = Array.from(toks);
    /* [v5.3.2] hidden templates should not contribute hard OOS signals */
    let cardRendered = false;
    try { cardRendered = card.getClientRects().length > 0; } catch (e) { /* ignore */ }
    function isRendered(el) {
      try { return el.getClientRects().length > 0 || getComputedStyle(el).display === 'contents'; }
      catch (e) { return true; }
    }
    for (let i = 0; i < els.length; i++) {
      const el = els[i];
      const r = classifyAttrs(el, !!(titleEl && el === titleEl));
      if (r.oos && cardRendered && !isRendered(el)) { r.oos = 0; r.why = ''; }
      if (r.oos) {
        ev.score += r.oos;
        if (r.oos >= 3) ev.hardOos = true;
        if (r.why && ev.why.length < 6) ev.why.push(r.why);
      }
      if (r.instock >= 3) ev.hardInStock = true;
      else if (r.instock === 2) ev.score -= 1;
    }
    const badge = scanTextBadges(card, titleEl);
    if (badge) { ev.score += 3; ev.hardOos = true; ev.why.push('badge "' + badge + '"'); }
    const buys = scanBuy(card);
    if (buys.oosLabel) { ev.score += 3; ev.hardOos = true; ev.why.push('add-to-cart says sold out'); }
    ev.feats.buyPresent = buys.present;
    ev.feats.buyCount = buys.count;
    ev.feats.buyOn = buys.on;
    ev.feats.buyAllDisabled = buys.present && !buys.oosLabel && buys.count > 0 && buys.on === 0;
    try {
      if (card.querySelector('input[name*="notify" i], input[name*="waitlist" i], input[name*="backorder" i], input[placeholder*="notify" i], form[action*="notify" i], [class*="waitlist" i]')) {
        ev.score += 3; ev.hardOos = true; ev.why.push('notify/waitlist widget');
      }
    } catch (e) { /* ignore */ }
    const f = ev.feats;
    try {
      const cs = getComputedStyle(card);
      f.opacity = parseFloat(cs.opacity);
      if (!isFinite(f.opacity)) f.opacity = 1;
      if (/grayscale\s*\(\s*(0?\.[5-9]|1)/i.test(cs.filter || cs.webkitFilter || '')) f.gray = true;
    } catch (e) { f.opacity = 1; }
    f.strike = !!qs(card, 's, del, [class*="strike" i], [class*="linethrough" i]');
    const pes = qsAll(card, PRICE_HINT).slice(0, 3);
    for (let i = 0; i < pes.length; i++) {
      try { if (/line-through/.test(getComputedStyle(pes[i]).textDecorationLine || '')) { f.strike = true; break; } } catch (e) { /* ignore */ }
    }
    f.hasPrice = PRICE_RE.test(textOf(card));
    return ev;
  }

  function evidenceFor(card, titleEl) {
    let e = OOS_CACHE.get(card);
    if (!e) { e = computeEvidence(card, titleEl); OOS_CACHE.set(card, e); }
    return e;
  }

  function bumpOos(card, titleEl, delta, why) {
    const e = evidenceFor(card, titleEl);
    e.score += delta;
    if (delta >= 3) e.hardOos = true;
    if (why && e.why.length < 6) e.why.push(why);
    return e;
  }

  function soldOutIn(card, titleEl) {
    try {
      const e = evidenceFor(card, titleEl);
      if (e.hardInStock) return false;
      return e.hardOos || e.score >= 2;
    } catch (err) {
      recordError('soldOutIn', err);
      console.error('[Harvest] OOS evidence crashed on card:', err, card);
      return false;
    }
  }

  function modelHit(m, tokens) {
    if (!m || !tokens || !tokens.length) return 0;
    for (let i = 0; i < tokens.length; i++) if (m.rules.indexOf(tokens[i]) !== -1) return 1;
    for (let i = 0; i < tokens.length; i++) if (m.neg.indexOf(tokens[i]) !== -1) return -1;
    return 0;
  }

  function learnModel(dom, prev) {
    const conf = [], ok = [];
    for (let i = 0; i < dom.length; i++) {
      const ev = dom[i].ev;
      if (ev.hardOos && !ev.hardInStock) conf.push(ev);
      else if (!ev.hardInStock && ev.score <= 1 && ev.feats.hasPrice) ok.push(ev);
    }
    if (conf.length < 2 || ok.length < 3) return prev || null;
    const total = dom.length;
    const freq = new Map(), oos = new Map(), ost = new Map();
    dom.forEach(x => x.ev.tokens.forEach(t => freq.set(t, (freq.get(t) || 0) + 1)));
    conf.forEach(ev => ev.tokens.forEach(t => oos.set(t, (oos.get(t) || 0) + 1)));
    ok.forEach(ev => ev.tokens.forEach(t => ost.set(t, (ost.get(t) || 0) + 1)));
    const rules = [], neg = [];
    oos.forEach((n, t) => {
      const oRate = n / conf.length;
      const iRate = (ost.get(t) || 0) / ok.length;
      if (oRate >= 0.6 && iRate <= 0.15 && (freq.get(t) || 0) <= total * 0.4) rules.push(t);
    });
    ost.forEach((n, t) => {
      const iRate = n / ok.length;
      if (iRate >= 0.6 && !oos.has(t) && (freq.get(t) || 0) <= total * 0.6) neg.push(t);
    });
    if (prev && prev.rules) {
      prev.rules.forEach(t => { if (rules.indexOf(t) === -1) rules.push(t); });
      prev.neg.forEach(t => { if (neg.indexOf(t) === -1) neg.push(t); });
    }
    const model = { host: host, rules: rules, neg: neg, learned: conf.length, clean: ok.length, total: total, at: Date.now() };
    if (rules.length) {
      let hit = 0;
      dom.forEach(x => { if (modelHit(model, x.ev.tokens) > 0) hit++; });
      console.log('[Harvest] sold-out model: ' + rules.length + ' rule(s) from ' +
        conf.length + ' sold-out / ' + ok.length + ' in-stock samples, covers ' + hit + '/' + total + ' cards',
        rules.slice(0, 12));
    }
    return model;
  }

  function loadOosModel() {
    try {
      const m = JSON.parse(sessionStorage.getItem(OOS_MODEL_KEY) || 'null');
      if (m && m.host === host && m.rules && m.rules.length) return m;
    } catch (e) { /* ignore */ }
    return null;
  }

  function saveOosModel(m) {
    try {
      if (m) sessionStorage.setItem(OOS_MODEL_KEY, JSON.stringify(m));
      else sessionStorage.removeItem(OOS_MODEL_KEY);
    } catch (e) { /* ignore */ }
  }

  function pageStats(dom) {
    const s = { n: dom.length, ops: [], strike: 0, okCount: 0, strikeOk: 0, price: 0, buyTotal: 0, buyEnabled: 0 };
    dom.forEach(x => {
      const f = x.ev.feats;
      if (isFinite(f.opacity)) s.ops.push(f.opacity);
      if (f.strike) s.strike++;
      if (f.hasPrice) s.price++;
      if (!x.ev.hardOos) {
        s.okCount++;
        if (f.strike) s.strikeOk++;
        if (f.buyPresent) { s.buyTotal++; if (f.buyOn > 0) s.buyEnabled++; }
      }
    });
    s.ops.sort((a, b) => a - b);
    s.medOpacity = s.ops.length ? s.ops[Math.floor(s.ops.length / 2)] : 1;
    s.strikeOkRate = s.okCount ? s.strikeOk / s.okCount : 0;
    s.priceRate = s.n ? s.price / s.n : 0;
    s.buyEnabledRate = s.buyTotal ? s.buyEnabled / s.buyTotal : 0;
    return s;
  }

  function relativeScore(ev, s) {
    if (!ev || !s || s.n < 5) return 0;
    let add = 0;
    if (s.medOpacity >= 0.95 && ev.feats.opacity <= 0.75) add += 1;
    if (s.strikeOkRate <= 0.25 && ev.feats.strike && !ev.feats.buyPresent) add += 1;
    if (s.buyTotal >= 3 && s.buyEnabledRate >= 0.5 && ev.feats.buyPresent && ev.feats.buyAllDisabled) add += 2;
    if (s.priceRate >= 0.75 && !ev.feats.hasPrice) add += 1;
    return add;
  }

  function ldTitleMap(ld) {
    const m = new Map();
    (ld || []).forEach(l => { if (l && l.oos) { const k = KEY(l.title); if (k.length > 6) m.set(k, true); } });
    return m.size ? m : null;
  }

  function decideOos(x, model, rel) {
    const ev = x.ev;
    if (!ev) return !!x.oos;
    if (ev.hardInStock) return false;
    if (ev.hardOos) return true;
    let score = ev.score + (rel || 0);
    const mh = modelHit(model, ev.tokens);
    if (mh > 0) score += 2; else if (mh < 0) score -= 1;
    return score >= 2;
  }

  function finalizePage(items, model, ld) {
    const raw = (items || []).map(it => ({
      title: it.title, price: it.price, oos: !!it.oos, url: it.url || '',
      rating: it.rating || '', reviews: it.reviews || '',
      card: it.card || null, titleEl: it.titleEl || null,
      ev: it.card ? safeCall(function () { return evidenceFor(it.card, it.titleEl); }, 'evidenceFor') : null
    }));
    const dom = raw.filter(x => x.ev);
    if (dom.length >= 5) {
      const m = learnModel(dom, model);
      if (m && m !== model) { model = m; saveOosModel(model); }
    }
    const forced = ldTitleMap(ld);
    if (forced) {
      dom.forEach(x => {
        const k = KEY(x.title);
        if (k && forced.has(k) && !x.ev.hardOos && !x.ev.hardInStock) { x.ev.hardOos = true; x.ev.why.push('json-ld availability'); }
      });
    }
    let activeModel = model;
    const soft = dom.filter(x => !x.ev.hardOos && !x.ev.hardInStock);
    if (model && soft.length >= 4) {
      const hits = soft.filter(x => modelHit(model, x.ev.tokens) > 0).length;
      if (hits / soft.length > 0.6) {
        activeModel = null;
        const strikes = (model.strikes || 0) + 1;
        if (strikes >= 2) {
          saveOosModel(null); model = null;
          console.log('[Harvest] learned model discarded — flagged the majority of cards on 2 pages');
        } else {
          model = Object.assign({}, model, { strikes: strikes });
          saveOosModel(model);
          console.log('[Harvest] learned model suspended this page (' + hits + '/' + soft.length + ' soft cards hit)');
        }
      } else if (model.strikes) {
        model = Object.assign({}, model, { strikes: 0 });
        saveOosModel(model);
      }
    }
    const stats = pageStats(dom);
    const out = raw.map(x => ({ title: x.title, price: x.price, oos: decideOos(x, activeModel, relativeScore(x.ev, stats)), url: x.url || '', rating: x.rating, reviews: x.reviews }));
    return { items: out, model: model };
  }

  function norm(s) { return String(s || '').replace(/\s+/g, ' ').trim(); }

  /* [v5.3.1] normalise Arabic digits + collapse split Arabic currencies */
  function prepPriceText(t) {
    return String(t || '')
      /* Arabic-Indic digits ٠-٩ (U+0660-U+0669) → ASCII */
      .replace(/[\u0660-\u0669]/g, function (d) { return String.fromCharCode(d.charCodeAt(0) - 0x0660 + 0x30); })
      /* Extended Arabic-Indic / Persian digits ۰-۹ (U+06F0-U+06F9) → ASCII */
      .replace(/[\u06F0-\u06F9]/g, function (d) { return String.fromCharCode(d.charCodeAt(0) - 0x06F0 + 0x30); })
      .replace(/\s+/g, ' ')
      .replace(/([A-Za-z])(\d)/g, '$1 $2')
      .replace(/(\d)([A-Za-z])/g, '$1 $2')
      .replace(DUP_CURRENCY, '$1')
      .replace(/(EGP|USD|SAR|AED|EUR|GBP|LE)(?=\s*\1)/gi, '')
      /* collapse "جنيه مصرى" to "جنيه" */
      .replace(/(جنيه|جنية)\s*(?:مصري|مصرى|مصريًا)/g, '$1')
      .trim();
  }
  function looksPrice(t) {
    const s = prepPriceText(norm(t));
    if (s.length < 2 || s.length > 80 || /%/.test(s) || /^[0-5][.,]\d$/.test(s)) return false;
    return PRICE_RE.test(s);
  }
  function cleanPrice(t) {
    const s = prepPriceText(norm(t));
    const m = s.match(PRICE_RE);
    const out = m ? norm(m[0]) : s.slice(0, 56);
    return out.replace(/(EGP|SAR|AED|USD)(?:\1)+/gi, '$1').replace(/(\d)([A-Za-z])/g, '$1 $2');
  }

    return false;
  }

  function itemUrl(it) {
    if (!it) return '';
    if (it.url) return it.url;
    const els = [it.titleEl, it.card];
    for (let i = 0; i < els.length; i++) {
      const el = els[i];
      if (!el || !el.tagName) continue;
      if (el.tagName === 'A' && el.href) return el.href;
      if (el.querySelector) {
        const a = el.querySelector('a[href]');
        if (a) return a.href;
      }
      if (el.closest) {
        const a = el.closest('a[href]');
        if (a) return a.href;
      }
    }
    return '';
  }
  function unique(items) {
    const map = new Map(); const out = [];
    items.forEach((it) => {
      const url = itemUrl(it);
      const k = url || it.title.toLowerCase();
      if (map.has(k)) {
        const p0 = map.get(k);
        if (!p0.rating && it.rating) p0.rating = it.rating;
        if (!p0.reviews && it.reviews) p0.reviews = it.reviews;
        if (it.oos) {
          const p = map.get(k);
          p.oos = true;
          const e = p.card && OOS_CACHE.get(p.card);
          if (e) e.hardOos = true;
        }
        return;
      }
      const c = { title: it.title, price: it.price, oos: !!it.oos, url: url, rating: it.rating || '', reviews: it.reviews || '', card: it.card || null, titleEl: it.titleEl || null };
      map.set(k, c); out.push(c);
    });
    return out;
  }

  function shopifyCardFor(links, handle) {
    let card = links[0];

    while (card && card !== document.body && card !== document.documentElement && guard++ < 12) {
      let foreign = false;
      let own = 0;
      const all = card.querySelectorAll('a[href*="/products/"]');
      for (let i = 0; i < all.length; i++) {
        const m = (all[i].getAttribute('href') || '').match(/\/products\/([^\/?#]+)/);
        if (!m) continue;
        if (m[1] === handle) own++;
        else { foreign = true; break; }
      }
      if (!foreign && own >= links.length) return card;
      card = card.parentElement;
    }
    let n = links[0];
    while (n && n !== document.body) {
      if (n.querySelector && n.querySelector('img')) return n;
      n = n.parentElement;
    }
    return links[0].parentElement;

  function relativeSelector(card, target) {
    if (!target || card === target) return ':scope';
    const tag = target.tagName.toLowerCase();
    const meaningful = meaningfulClasses(target);
    if (meaningful.length) {
      const sel = tag + '.' + meaningful.slice(0, 2).map(cssEscape).join('.');
      if (qsAll(card, sel).length <= 3) return sel;
    }
    const itemprop = target.getAttribute('itemprop');
    if (itemprop) return '[itemprop="' + cssEscape(itemprop) + '"]';
    const dataKeys = ['data-price', 'data-price-amount', 'data-qa'];
    for (let i = 0; i < dataKeys.length; i++) {
      if (target.getAttribute(dataKeys[i]) != null) {
        const v = target.getAttribute(dataKeys[i]);
        return v ? '[' + dataKeys[i] + '="' + cssEscape(v) + '"]' : '[' + dataKeys[i] + ']';
      }
    }
    if (tag === 'img') return 'img[alt]';
    if (/^h[1-6]$/.test(tag)) {
      const withA = target.querySelector('a') ? tag + ' a' : tag;
      if (qsAll(card, withA).length <= 3) return withA;
    }
    if (target.getAttribute('href')) return 'a[href]';
    const parent = target.parentElement;
    if (parent && parent !== card) {
      const parentSel = relativeSelector(card, parent);
      const kids = Array.from(parent.children).filter((c) => c.tagName === target.tagName);
      const idx = kids.indexOf(target);
      if (idx >= 0 && kids.length > 1) return parentSel + ' > ' + tag + ':nth-of-type(' + (idx + 1) + ')';
      return parentSel + ' > ' + tag;
    }
    return tag;
  }

  function clusterFromSource() {
    const priceEls = [];
    qsAll(document, PRICE_HINT).forEach((el) => { if (looksPrice(textOf(el))) priceEls.push(el); });
    if (priceEls.length < 2) {
      document.querySelectorAll('span, div, p, b, strong, em, h3, h4, h5').forEach((el) => {
        if (el.childElementCount > 3) return;
        const t = textOf(el);
        if (t.length <= 56 && looksPrice(t)) priceEls.push(el);
      });
    }
    const clusters = new Map();
    function consider(cards, signature, score) {
      /* [v5.3] filter composite rows */
      cards = cards.filter((c) => !isCompositeCard(c));
      if (cards.length < 2) return;
      if (HIJACK_RE.test(signature)) return;
      if (inHijackZone(cards[0]) || inBoilerplate(cards[0])) return;
      const key = signature + ':' + cards.length;
      const prev = clusters.get(key);
      if (!prev || score > prev.score) clusters.set(key, { cards, score, signature });
    }
    function clusterWalk(anchors, relevant) {
      anchors.forEach((anchor) => {
        let node = anchor;
        let depth = 0;
        while (node && node !== document.body && depth < 10) {
          const parent = node.parentElement;
          if (!parent) break;
          const sig = classSig(node);
          const siblings = Array.from(parent.children).filter((c) => c.tagName === node.tagName && classSig(c) === sig);
          if (siblings.length >= 2) {
            const good = siblings.filter(relevant);
            if (good.length >= 2) {
              const score = good.length * 4 + good.filter((c) => c.querySelector('img')).length * 2 + good.filter((c) => c.querySelector('a[href]')).length * 2 - depth;
              consider(good, sig, score);
            }
          }
          node = parent;
          depth += 1;
        }
      });
    }
    clusterWalk(priceEls, (s) => PRICE_RE.test(textOf(s)) || !!qs(s, PRICE_HINT));
    const productLinks = qsAll(document, 'a[href*="/products/"], a[href*="/product/"], a[href*="/shop/"]')

  function deriveSelectors(cards) {
    const card = sharedCardSelector(cards);
    const titleSels = []; const priceSels = []; const linkSels = []; const imageSels = []; const oosSels = [];
    cards.forEach((c) => {
      const titleEl = findTitleEl(c);
      if (titleEl) titleSels.push(relativeSelector(c, titleEl));
      const priceEl = findPriceEl(c);
      if (priceEl) priceSels.push(relativeSelector(c, priceEl));
      if (c.matches && c.matches('a[href]')) linkSels.push(':scope');
      else {
        const a = c.querySelector('a[href]');
        if (a) {
          const m = meaningfulClasses(a);
          linkSels.push(m.length ? 'a.' + m.slice(0, 2).map(cssEscape).join('.') : 'a[href]');
        }
      }
      if (c.querySelector('img[alt]')) imageSels.push('img[alt]');
      const oos = qs(c, '[class*="sold-out" i], [class*="sold_out" i], [class*="soldout" i], [class*="out-of-stock" i], [class*="out_of_stock" i], [class*="outofstock" i], [class*="unavailable" i], [class~="oos" i], [data-available="false"], [data-sold-out="true"]');
      if (oos) oosSels.push(relativeSelector(c, oos));
    });
    const nextSel = ['a[rel="next"]', 'a.pagination__next', 'a.next.page-numbers', 'a.s-pagination-next', 'a.next'].find((s) => qs(document, s));
    return {
      card,
      title: mostCommon(titleSels) || 'h2, h3, a[href]',
      price: mostCommon(priceSels) || '[class*="price" i], [itemprop="price"]',
      link: mostCommon(linkSels),
      image: mostCommon(imageSels),
      oos: mostCommon(oosSels),
      next: nextSel
    };
  }
  function applySelectors(selectors) {
    if (!selectors || !selectors.card) return [];
    const items = [];
    let skippedZone = 0;
    let skippedComposite = 0;
    qsAll(document, selectors.card).forEach((card) => {
      if (inHijackZone(card) || inBoilerplate(card)) { skippedZone++; return; }
      /* [v5.3] */
      if (isCompositeCard(card)) { skippedComposite++; return; }
      let titleEl = selectors.title ? qs(card, selectors.title) : findTitleEl(card);
      let title = textOf(titleEl);
      let isImgTitle = !!(titleEl && titleEl.tagName === 'IMG');
      if (junkTitle(title) && selectors.image) {
        const img = qs(card, selectors.image);
        title = img ? norm(img.getAttribute('alt')) : title;
        if (!junkTitle(title)) { titleEl = img || titleEl; isImgTitle = true; }
      }
      if (junkTitle(title)) {
        const img = card.querySelector('img[alt]');
        title = img ? norm(img.getAttribute('alt')) : title;
        if (img && !junkTitle(title)) { titleEl = img; isImgTitle = true; }
      }
      if (junkTitle(title)) return;
      let price = 'N/A';
      const pEl = selectors.price ? qs(card, selectors.price) : findPriceEl(card);
      if (pEl) {
        const t = textOf(pEl);
        if (looksPrice(t)) price = cleanPrice(t);
        else {
          const attr = pEl.getAttribute('content') || pEl.getAttribute('data-price') || pEl.getAttribute('data-price-amount');
          if (attr && looksPrice(attr)) price = cleanPrice(attr);
          else if (t) price = cleanPrice(t);
        }
      }
      try {
        evidenceFor(card, titleEl);
        if (selectors.oos && qs(card, selectors.oos)) bumpOos(card, titleEl, 3, 'compiled oos selector');
      } catch (err) {
        recordError('applySelectors (evidence)', err);
        console.error('[Harvest] evidence crashed on card:', err);
      }
      items.push({ title, price, oos: soldOutIn(card, titleEl), card, titleEl });
    });
    if (skippedZone) console.log('[Harvest] skipped ' + skippedZone + ' card(s) in recommendation/boilerplate zone');
    if (skippedComposite) console.log('[Harvest] skipped ' + skippedComposite + ' composite row(s)');
    return unique(items);
  }
  function ldTitleOverlap(applied, jsonld) {
    const ldKeys = jsonld.map((j) => KEY(j.title)).filter((k) => k.length > 3);
    if (!ldKeys.length) return true;
    return applied.some((it) => {
      const k = KEY(it.title);
      if (k.length <= 3) return false;
      return ldKeys.some((lk) => lk === k ||
        (Math.min(lk.length, k.length) >= 10 && (lk.indexOf(k) === 0 || k.indexOf(lk) === 0)));
    });
  }
  function compileFromSource() {
    const ranked = clusterFromSource();
    const jsonld = scrapeJsonLd();

      };
    }

    const applied = applySelectors(selectors);
    console.log('[Harvest] compiled cluster:', best.signature, '×', best.cards.length);
    if (applied.length && jsonld.length >= 3 && !ldTitleOverlap(applied, jsonld)) {

      host,
      selectors,
      notes: 'Compiled from source · ' + best.signature + ' × ' + best.cards.length + ' · card `' + selectors.card + '`'
    };

      return { items: jsonld, blueprint: { method: 'json-ld', host, selectors: { card: '', title: 'Product.name', price: 'Offer.price' }, notes: 'JSON-LD Product nodes' } };
    }
    return { items: applied, blueprint };
  }

    };

        if (reused && reused.length >= items.length) {
          items = reused; blueprint = saved; platform = 'compiled';
        }
      }
    }
    if (items.length < 2) {
      const compiled = safeCall(compileFromSource, 'compileFromSource') || { items: [] };
      if ((compiled.items || []).length > items.length) {
        items = compiled.items;
        blueprint = compiled.blueprint;
        platform = compiled.blueprint && compiled.blueprint.method === 'json-ld' ? 'json-ld' : 'compiled';

      }

  function isDisabledNav(el) {
    if (!el) return true;
    if (el.getAttribute && el.getAttribute('aria-disabled') === 'true') return true;
    if (el.disabled) return true;
    if (el.classList && (el.classList.contains('disabled') || el.classList.contains('s-pagination-disabled') || el.classList.contains('ais-Pagination-item--disabled'))) return true;
    if (el.closest && el.closest('li.disabled, li.page-item.disabled, .ais-Pagination-item--disabled')) return true;
    return false;
  }
  function pageNumFromHref(href) {
    if (!href || href === '#' || /^(javascript:|mailto:|tel:)/i.test(href)) return null;
    try {
      const u = new URL(href, location.href);
      const keys = ['page', 'paged', 'product-page', 'pg', 'page_number'];
      for (let i = 0; i < keys.length; i++) {
        const q = u.searchParams.get(keys[i]);
        if (q && /^\d+$/.test(q)) return parseInt(q, 10);
      }
      const m = u.pathname.match(/\/page\/(\d+)/i);
      if (m) return parseInt(m[1], 10);
      const p = u.searchParams.get('p');
      if (p && /^\d+$/.test(p)) {
        const n = parseInt(p, 10);
        if (n >= 1 && n < 500) return n;
      }
    } catch (e) { /* ignore */ }
    return null;
  }
  function resolvePagerHref(href) {
    if (!href || href === '#' || /^(javascript:|mailto:|tel:)/i.test(href)) return null;
    try {
      const u = new URL(href, location.href);
      if (u.protocol === 'http:' || u.protocol === 'https:') {
        const keys = ['page', 'paged', 'product-page', 'pg', 'page_number', 'p'];
        const hasPage = keys.some(k => u.searchParams.has(k)) || /\/page\/\d+/i.test(u.pathname);
        if (hasPage) return u.href;

        const cur = new URL(location.href);
        u.searchParams.forEach((v, k) => {
          if (!cur.searchParams.has(k)) cur.searchParams.set(k, v);
        });
        return cur.href;
      }
    } catch (e) { /* ignore */ }
    return href;
  }
  function currentPageNum() {
    const fromUrl = pageNumFromHref(location.href);
    if (fromUrl) return fromUrl;
    const active = document.querySelector('.page-item.active, li.active, [aria-current="page"], .s-pagination-selected, .current, .ais-Pagination-item--selected');
    if (active) {
      const t = parseInt(textOf(active), 10);
      if (!isNaN(t) && t > 0 && t < 500) return t;
    }
    return pageNumFromHref(location.href) || 1;
  }
  function pagerNodes() {
    const sel = '.pagination, ul.pagination, nav.pagination, .yalla-pagination, .s-pagination-strip, nav[aria-label*="pagination" i], .woocommerce-pagination, .pager, .pages, [data-scope="pagination"], .paginate, .wp-pagenavi, .ais-Pagination';

    roots.forEach((root) => {

    const LOAD_MORE_RE = /(?:load|view|show|see)\s*more|عرض\s*(?:المزيد|اكثر|أكثر)|مشاهدة\s*المزيد|إظهار\s*المزيد|اظهار\s*المزيد|تحميل\s*المزيد|المزيد\s*من\s*(?:المنتجات|الأجهزة|العروض)/i;

    const candidateNodes = document.querySelectorAll('button, a, [role="button"], [class*="btn" i], [class*="button" i], [class*="more" i], div[tabindex], span[tabindex]');
    for (let i = 0; i < candidateNodes.length; i++) {
      const el = candidateNodes[i];
      if (isDisabledNav(el) || inBoilerplate(el) || inHijackZone(el) || inFilterZone(el) || isInsideCard(el) || inChrome(el)) continue;
      const t = norm(el.textContent);
      if (!t || t.length > 70) continue;
      /* [v5.3] skip bare "more"/"المزيد" */
      if (BARE_MORE.test(t)) continue;
      if (LOAD_MORE_RE.test(t)) {
        if (!isVisible(el)) continue;
        if (looksPrice(t)) continue;
        return el;
      }
    }

    const textEls = document.querySelectorAll('span, p, b, strong');
    for (let i = 0; i < textEls.length; i++) {
      const s = textEls[i];
      if (s.children.length > 2) continue;
      const t = norm(s.textContent);
      if (BARE_MORE.test(t)) continue;
      if (t && t.length <= 40 && LOAD_MORE_RE.test(t)) {
        const btn = s.closest('button, a, [role="button"], [class*="btn" i], [class*="button" i]') || s;
        if (btn && !isDisabledNav(btn) && !inBoilerplate(btn) && !inHijackZone(btn) && !inFilterZone(btn) && !isInsideCard(btn) && !inChrome(btn)) {
          if (!isVisible(btn)) continue;
          if (looksPrice(norm(btn.textContent))) continue;
          return btn;
        }
      }
    }

    return null;
  }

  function triggerClick(el) {
    if (!el) return;
    try { el.scrollIntoView({ block: 'center' }); } catch (e) { /* ignore */ }
    const target = el.querySelector('button, a, [role="button"]') || el;
    const fire = (node) => {
      if (!node) return;
      const opts = { bubbles: true, cancelable: true, view: window };
      ['pointerdown', 'mousedown', 'pointerup', 'mouseup', 'click'].forEach((evt) => {
        try { node.dispatchEvent(new MouseEvent(evt, opts)); } catch (e) { /* ignore */ }
      });
      try { node.click(); } catch (e) { /* ignore */ }
    };
    fire(target);
    if (target !== el) fire(el);
    setTimeout(() => {
      fire(target);
      if (target !== el) fire(el);
    }, 150);
  }

  function mergeItems(existing, incoming) {
    const seen = new Map();
    const out = [];
    (existing || []).concat(incoming || []).forEach((it) => {
      if (!it || !it.title) return;
      const k = it.url || it.title.toLowerCase();
      if (seen.has(k)) {
        const p = seen.get(k);
        if (it.oos) p.oos = true;
        if (!p.rating && it.rating) p.rating = it.rating;
        if (!p.reviews && it.reviews) p.reviews = it.reviews;
        return;
      }
      const c = { title: it.title, price: it.price, oos: !!it.oos, url: it.url || '', rating: it.rating || '', reviews: it.reviews || '' };
      seen.set(k, c);
      out.push(c);
    });
    return out;
  }

  function exportToFile(items) {
    if (!items || !items.length) { setStatus('Nothing to export (all sold out?).'); return; }
    const tag = safeCall(detectPlatform, 'detectPlatform') || 'unknown';
    const state = loadState();

    let siteName = host.replace(/^www\./i, '');
    try { if (sourceUrl) siteName = new URL(sourceUrl).hostname.replace(/^www\./i, ''); } catch (e) { /* ignore */ }
    const safeSiteName = siteName.replace(/[^a-z0-9.-]/gi, '_');
    const bp = state.blueprint || loadBlueprint();
    const model = loadOosModel();
    let header = '# ' + (state.platform || tag) + ' listings' + (INCLUDE_OOS ? '' : ' (in stock only)') + '\n\n**Source URL:** ' + sourceUrl + '\n\n';
    if (bp && bp.selectors && bp.selectors.card) {
      header += '**Extractor compiled from page source**\n\n```\ncard    ' + bp.selectors.card + '\ntitle   ' + bp.selectors.title + '\nprice   ' + bp.selectors.price + '\n```\n\n';
    }
    if (model && model.rules && model.rules.length) {
      header += '**Learned sold-out markers:** ' + model.rules.slice(0, 10).map((r) => '`' + r + '`').join(' ') + '\n\n';
    }
    const withOos = items.some((it) => it.oos);
    const withRev = items.some((it) => it.rating || it.reviews);
    const cols = ['#', 'Title', 'Price'].concat(withRev ? ['Rating', 'Reviews'] : [], withOos ? ['Stock'] : []);
    header += '| ' + cols.join(' | ') + ' |\n|' + cols.map(() => '---').join('|') + '|\n';
    const rows = items.map((it, i) => {
      const c = [i + 1, escapeMd(it.title), escapeMd(it.price)];

      if (withOos) c.push(it.oos ? 'sold out' : 'in stock');
      return '| ' + c.join(' | ') + ' |';
    });
    const text = header + rows.join('\n') + '\n';
    const blob = new Blob([text], { type: 'text/markdown;charset=utf-8' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');

  }

  function makeDraggable(panel, handle) {
    let dragging = false, offsetX = 0, offsetY = 0;
    handle.addEventListener('pointerdown', (e) => {
      if (e.target.closest('button')) return;
      dragging = true;
      const rect = panel.getBoundingClientRect();
      offsetX = e.clientX - rect.left;
      offsetY = e.clientY - rect.top;
      panel.style.setProperty('left', rect.left + 'px', 'important');
      panel.style.setProperty('top', rect.top + 'px', 'important');
      panel.style.setProperty('right', 'auto', 'important');
      handle.setPointerCapture(e.pointerId);
      e.preventDefault();
    });
    handle.addEventListener('pointermove', (e) => {
      if (!dragging) return;
      e.preventDefault();
      const maxLeft = Math.max(0, window.innerWidth - panel.offsetWidth);
      const maxTop = Math.max(0, window.innerHeight - panel.offsetHeight);
      panel.style.setProperty('left', Math.min(Math.max(0, e.clientX - offsetX), maxLeft) + 'px', 'important');
      panel.style.setProperty('top', Math.min(Math.max(0, e.clientY - offsetY), maxTop) + 'px', 'important');
    });
    handle.addEventListener('pointerup', (e) => {
      dragging = false;
      if (handle.hasPointerCapture && handle.hasPointerCapture(e.pointerId)) handle.releasePointerCapture(e.pointerId);
    });
    handle.addEventListener('pointercancel', () => { dragging = false; });
  }

  function runStep() {
    const myGen = stepGen;
    if (!loadState().running) return;
    /* [v5.3] Algolia / Journal3 IAS quiet mode: don't thrash the page */
    const quiet = isAlgoliaSite() || isIasSite();

      try { window.scrollTo(0, document.body.scrollHeight); } catch (e) { /* ignore */ }

    setTimeout(() => {
      if (stepGen !== myGen) return;
      if (!quiet) {
        try { window.scrollTo(0, 0); } catch (e) { /* ignore */ }

  }

  function collectAndAdvance() {
    const state = loadState();
    if (!state.running) { finishRun(state.items, 'Stopped.'); return; }
    const preview = scrapeCurrentPage();
    const before = (state.items || []).length;
    const merged = mergeItems(state.items, preview.items);
    const pn = currentPageNum();
    const scrapedPages = (state.scrapedPages || []).slice();
    if (scrapedPages.indexOf(pn) === -1) scrapedPages.push(pn);
    const stall = merged.length === before ? (state.stall || 0) + 1 : 0;
    const fresh = Object.assign({}, state, {
      items: merged,
      skipped: (state.skipped || 0) + preview.skipped,
      lastScrapedPage: pn,
      scrapedPages: scrapedPages,
      stall: stall
    });
    saveState(fresh);
    setStatus('Page ' + pn + ' · +' + preview.items.length + ' here · ' + merged.length + ' total' +
      (preview.skipped ? ' · ' + preview.skipped + ' sold out ignored' : '') + ' · ' + preview.platform);
    advance(fresh, merged);
  }

  function advance(fresh, merged) {
    const myGen = stepGen;
    if ((fresh.stall || 0) >= 3) { finishRun(merged, 'No new items — reached the end. Exported ' + merged.length + '.'); return; }
    const scrapedCount = Math.max((fresh.scrapedPages && fresh.scrapedPages.length) || 0, (fresh.loadMoreClicks || 0) + 1);
    if (scrapedCount >= fresh.maxPages) { finishRun(merged, 'Page cap reached (' + fresh.maxPages + '). Exported ' + merged.length + '.'); return; }

    const more = findLoadMore();
    if (more) {
      fresh.loadMoreClicks = (fresh.loadMoreClicks || 0) + 1;
      saveState(fresh);
      setStatus('Loading more items… (pass ' + fresh.loadMoreClicks + ')');
      triggerClick(more);
      setTimeout(() => { if (stepGen === myGen) runStep(); }, PAGE_LOAD_WAIT_MS);
      return;
    }

    const btn = findNextButton();
    if (btn) {
      triggerClick(btn);
      setTimeout(() => { if (stepGen === myGen) runStep(); }, PAGE_LOAD_WAIT_MS);
      return;
    }

    if (/(^|\.)noon\.com$/.test(host)) {
      const nextA = findNextAnchor();
      if (nextA) {
        triggerClick(nextA);
        setTimeout(() => { if (stepGen === myGen) runStep(); }, PAGE_LOAD_WAIT_MS);
        return;
      }
    }

    const href = findNextHref();
    if (href) { location.href = href; return; }

    /* [v5.3] Algolia / Journal3 IAS sites: stop cleanly rather than infinite scroll */
    if (isAlgoliaSite() || isIasSite()) {
      finishRun(merged, 'No next page. Exported ' + merged.length + '.');
      return;
    }

    /* infinite-scroll fallback: scroll to bottom, wait, re-evaluate */
    try { window.scrollTo(0, document.body.scrollHeight); } catch (e) { /* ignore */ }
    setTimeout(() => {
      try { window.scrollBy(0, -500); } catch (e) { /* ignore */ }
      setTimeout(() => {
        try { window.scrollTo(0, document.body.scrollHeight); } catch (e) { /* ignore */ }
      }, 300);
    }, 600);
    setTimeout(() => { if (stepGen === myGen) collectAndAdvance(); }, PAGE_LOAD_WAIT_MS + 1500);
  }
